function petMove(dx, dy, k, ms) {
  const g = document.querySelector('.petg');
  g.style.transition = ms ? `transform ${ms}ms ease-in-out` : 'none';
  g.style.transformOrigin = `${RUG.cx}px ${RUG.bottom}px`;
  g.style.transform = `translate(${dx}px, ${dy}px) scale(${k})`;
}
function svgPoint(e) {
  const p = room.createSVGPoint(); p.x = e.clientX; p.y = e.clientY;
  return p.matrixTransform(room.getScreenCTM().inverse());
}
let drag = null;
room.addEventListener('pointerdown', e => {
  const g = e.target.closest('.petg.drag');
  if (!g || busy) return;
  drag = { start: svgPoint(e), moved: false, dx: 0, dy: 0 }; g.classList.add('dragging');
  room.setPointerCapture(e.pointerId);
});
room.addEventListener('pointermove', e => {
  if (!drag) return;
  const p = svgPoint(e), dx = p.x - drag.start.x, dy = p.y - drag.start.y;
  if (Math.hypot(dx, dy) > 8) drag.moved = true;
  petMove(dx, dy, dragK(dx, dy), 0);
  drag.dx = dx; drag.dy = dy; drag.p = p;
  dropHint(petDropSpot(dx, dy, p));   // the spot glows brighter while the pet is right on it; let go to drop
});
function dragK(dx, dy) {   // 1 on the rug, the spot's size once the feet reach the spot
  const t = TASKS[task], to = t && t.icon === 'scale' ? ON_SCALE : t && (t.icon === 'bubbles' || t.icon === 'brush') ? IN_TUB : null;
  if (!to) return 1;
  const all = Math.hypot(to.cx - RUG.cx, to.bottom - RUG.bottom), left = Math.hypot(to.cx - RUG.cx - dx, to.bottom - RUG.bottom - dy);
  return to.k + (1 - to.k) * Math.min(1, left / all);
}
function dropHint(id) { room.querySelectorAll('.tgt').forEach(g => g.classList.toggle('over', g.dataset.t === id)); }
// Drag and drop rule for the whole game (Mac, Nov 23): nothing pulls or jumps while you drag.
// The spot glows brighter when the thing is right on it, and it only counts when you let go there;
// then it glides gently into place.
const on = (t, x, y, below = 0) => x > t.x && x < t.x + t.w && y > t.y && y < t.y + t.h + below;
function petDropSpot(dx, dy, p) {
  const fx = RUG.cx + dx, fy = RUG.bottom + dy, t = TASKS[task];   // where the pet's feet are
  if (t && t.icon === 'scale' && on(TARGETS.scale, fx, fy, 25)) return 'scale';
  if (t && (t.icon === 'bubbles' || t.icon === 'brush') && (on(TARGETS.bath, fx, fy) || (p && on(TARGETS.bath, p.x, p.y)))) return 'bath';   // anywhere in the glowing tub box
  return null;
}
room.addEventListener('pointerup', () => {
  if (!drag) return;
  const d = drag; drag = null;
  document.querySelector('.petg').classList.remove('dragging'); dropHint(null);
  if (!d.moved) return;
  const spot = petDropSpot(d.dx, d.dy, d.p);
  if (spot === 'scale') weigh();
  else if (spot === 'bath') goToBath();
  else petMove(0, 0, 1, 450);   // not on the right spot: back to the rug, full size
});

const cupY = c => CUP.y0 - (CUP.y0 - CUP.y1) * c;
function cupInside(c) {   // the glass's inside from the bottom up to c cups (it is a little wider at the top)
  const y = cupY(c), t = (CUP.y0 - y) / (CUP.y0 - CUP.y1);
  const l = CUP.left0 + (CUP.left1 - CUP.left0) * t, r = CUP.right0 + (CUP.right1 - CUP.right0) * t;
  return `M${CUP.left0},${CUP.y0 - 10} Q${CUP.left0 + 8},${CUP.y0 + 4} ${(CUP.left0 + CUP.right0) / 2},${CUP.y0 + 5} Q${CUP.right0 - 8},${CUP.y0 + 4} ${CUP.right0},${CUP.y0 - 10} L${r},${y} Q${(l + r) / 2},${y - 6} ${l},${y} Z`;
}
function kibbleDots(c) {
  let s = '';
  if (c <= 0) return s;
  const top = cupY(c);
  for (let i = 0; i < 26; i++) {
    const y = CUP.y0 - 10 - ((i * 37) % 100) / 100 * (CUP.y0 - 10 - top), x = CUP.left0 + 16 + ((i * 53) % (CUP.right0 - CUP.left0 - 32));
    if (y > top + 6) s += `<circle cx="${x}" cy="${y}" r="5" fill="#a86d43"/>`;
  }
  return s;
}

  const st = document.getElementById('msStage'), from = (st.getAttribute('viewBox') || '').split(' ').map(Number);

function startPour(e) {
  if (!ms || ms.step < 1 || ms.done || ms.pouring) return;
  if (e) e.preventDefault();
  ms.pouring = true; ms.last = performance.now() + 550;   // the food starts once the bin is over the cup
  ready(false);
  document.getElementById('msBin').classList.add('pour');
  document.getElementById('msStage').classList.add('pouring');
  requestAnimationFrame(pourTick);
}
function pourTick(t) {
  if (!ms || !ms.pouring) return;
  if (t > ms.last) { setCup(ms.amount + POUR_RATE * (t - ms.last) / 1000); ms.last = t; }
  requestAnimationFrame(pourTick);
}
function stopPour() {
  if (!ms || !ms.pouring) return;
  ms.pouring = false;
  document.getElementById('msBin').classList.remove('pour');
  document.getElementById('msStage').classList.remove('pouring');
  judge();
}
// is the food at the line? Only then does the bowl show up.
function judge() {
  const goal = CUPS[pet.food], diff = ms.amount - pet.food;
  if (ms.amount <= 0) { ready(false); msButtons(null); msSay(`Hold the food bin to pour, and let go at the ${goal} line.`); return; }
  if (Math.abs(diff) <= CLOSE_ENOUGH) {
    ready(true); msButtons(null);
    msSay(`Just right! Now drag the cup to the bowl and hold it there to pour.`);
  } else if (diff < 0) {
    ready(false); msButtons(null);   // no spoon when there is too little
    msSay(`A little more. Pour up to the ${goal} line.`);
  } else {
    ready(false); msButtons(null, 'on');
    msSay(`Too much. Take some out with the spoon.`);
  }
}

const PAWS = {};   // filled in by build.py: each pose picture's paw width and paw middle, in picture pixels
function poseBox(pose, cx, bottom, sitH, byCollar) {
  const P = POSES[pose], k = sitH / 560, a = PAWS[P.art(pet)], s = PAWS[pet.art];
  if (!a || !s) return { x: cx - P.w * k / 2 + P.dx * k, y: bottom - P.h * k, w: P.w * k, h: P.h * k };
  // every pose is drawn so its paws are as wide as when sitting
  const f = s[0] / a[0], sx = cx - POSES.sit.w * k / 2, sy = bottom - 560 * k;   // where the sitting picture's corner is
  // in the tub (paws hidden) the collar stays in the same spot, so the head does not jump; on the floor the paws do
  if (byCollar) return { x: sx + s[2] * k - a[2] * k * f, y: sy + s[3] * k - a[3] * k * f, w: P.w * k * f, h: P.h * k * f };
  const mid = sx + s[1] * k;
  return { x: mid - a[1] * k * f, y: bottom - P.h * k * f, w: P.w * k * f, h: P.h * k * f };
}

// which parts of the pet can be rubbed: a grid over the picture, keeping only squares where the pet is
function petCells(artKey, box, cols, rows) {
  return new Promise(res => {
    const im = new Image();
    im.onload = () => {
      const c = document.createElement('canvas'); c.width = 60; c.height = 76;
      const g = c.getContext('2d'); g.drawImage(im, 0, 0, 60, 76);
      const d = g.getImageData(0, 0, 60, 76).data, cells = [];
      for (let r = 0; r < rows; r++) for (let q = 0; q < cols; q++) {
        const px = Math.floor((q + .5) / cols * 60), py = Math.floor((r + .5) / rows * 76);
        if (d[(py * 60 + px) * 4 + 3] > 200 && r < rows * .75)   // the bottom part is inside the tub
          cells.push({ x: box.x + (q + .5) / cols * box.w, y: box.y + (r + .5) / rows * box.h, done: false });
      }
      res(cells);
    };
    im.src = ART[artKey];
  });
}

  const s = STEPS()[bt.step], next = STEPS()[bt.step + 1];
  document.getElementById('btSteps').innerHTML = bathStepsHtml();
  document.getElementById('btSay').textContent = !s ? '' : bt.ready ? s.ok(pet.name) : s.say(pet.name);
  document.querySelectorAll('.bt-tool').forEach(g => {
    const i = STEPS().findIndex(x => x.id === g.dataset.tool);
    g.classList.toggle('on', !!s && ((i === bt.step && !bt.ready) || (bt.ready && i === bt.step + 1)));
    g.classList.toggle('used', i < bt.step);
  });
}
function nextBathStep() {
  if (!bt.ready || bt.step >= STEPS().length - 1) return;
  bt.step++; bt.ready = false;
  bt.cells.forEach(c => c.done = false);
  document.querySelectorAll('#btFx .puff').forEach(p => p.remove());
  if (STEPS()[bt.step].id === 'nails') nailSpots();
  if (STEPS()[bt.step].id === 'towel') document.querySelectorAll('#btFx .bub').forEach(b => b.remove());
  bathStep();
}

function dropTool() {
  if (!bt || !bt.drag) return;
  const d = bt.drag; bt.drag = null;
  if (bt.brushEnd && STEPS()[bt.step] && STEPS()[bt.step].id === 'brush' && bt.ready) { clearTimeout(bt.brushEnd); bt.brushEnd = null; setTimeout(bathDone, 200); }
  if (d.head) {
    document.getElementById('btHead').classList.remove('raining');
    document.querySelector('#btHead .ring').style.display = '';
    return setHead(...SHOWER_HOME);   // back on its hook
  }
  d.img.setAttribute('x', d.home[0]); d.img.setAttribute('y', d.home[1]); d.img.removeAttribute('transform');
  d.el.querySelector('.ring').style.display = '';
}

function rubAt(x, y, reach) {
  const step = STEPS()[bt.step].id, fx = document.getElementById('btFx');
  let hit = false;
  bt.cells.forEach((c, i) => {
    if (c.done || Math.hypot(c.x - x, c.y - y) > reach) return;
    c.done = true; hit = true;
    if (step === 'soap') {
      const jx = (i * 37) % 23 - 11, jy = (i * 53) % 19 - 9, r = 15 + (i * 7) % 9;   // a little uneven, like real bubbles
      fx.insertAdjacentHTML('beforeend', `<g class="bub" data-cell="${i}"><circle cx="${c.x - 8 + jx}" cy="${c.y + jy}" r="${r}" fill="#fff" opacity=".92"/><circle cx="${c.x + 12 + jx}" cy="${c.y - 12 + jy}" r="${r * .6}" fill="#fff" opacity=".92"/><circle cx="${c.x - 12 + jx}" cy="${c.y - 6 + jy}" r="4" fill="#d7eefc"/></g>`);
    } else if (step === 'nails') {   // snip: the paw's glow goes and a little star pops
      fx.querySelectorAll(`.nail-spot[data-cell="${i}"]`).forEach(n => n.remove());
      fx.insertAdjacentHTML('beforeend', `<polygon class="snip" points="${starPts(c.x, c.y - 6, 26, 11)}" fill="#ffd34d"/>`);
    } else if (step === 'rinse') {
      fx.querySelectorAll(`.bub[data-cell="${i}"]`).forEach(b => { b.style.opacity = 0; setTimeout(() => b.remove(), 400); });
    } else {
      const col = step === 'towel' ? '#bae5ca' : '#fff6b0';
      fx.insertAdjacentHTML('beforeend', `<circle class="puff" cx="${c.x}" cy="${c.y}" r="16" fill="${col}"/>`);
    }
  });
  if (!hit) return;
  const part = bt.cells.filter(c => c.done).length / bt.cells.length;
  if (step === 'soap') { const st = document.getElementById('btStink'); if (st) st.style.opacity = Math.max(0, 1 - part / RUB_DONE); }
  if (!bt.ready && part >= (step === 'nails' ? 1 : RUB_DONE)) stepReady(step);
}

function stepReady(step) {
  bt.ready = true;
  if (step === 'soap') { const st = document.getElementById('btStink'); if (st) st.remove(); }
  if (step === 'rinse') setDogPose('bath');
  if (step === 'towel') setDogPose('sit');
  if (step === 'nails') return groomDone();
  if (step === 'brush' && kind === 'cat') { setDogPose('bath'); return bathStep(); }   // happy and fluffy; the clippers blink next
  if (step === 'brush') { bt.brushEnd = setTimeout(() => { bt.brushEnd = null; dropTool(); bathDone(); }, 3000); return bathStep(); }
  bathStep();
}

function cfArrow(from, to) {   // the dashed arrow that shows where to drag next

}

const cfSrcKey = () => cf.phase === 'scoopDry' ? 'dry' : cf.phase === 'scoopWet' ? 'wet' : null;
function cfUpdate() {   // the checklist, which things glow, and where the arrow points
  const c = cf, picking = c.phase === 'pickDry' || c.phase === 'pickWet', k = cfSrcKey();
  document.getElementById('cfList').innerHTML = cfListSvg();
  document.querySelectorAll('#cfList .cf-amt').forEach(t => t.nextElementSibling.setAttribute('transform', `translate(${96 + t.getComputedTextLength() + 13} 0)`));
  document.querySelectorAll('.cf-tool').forEach(g => {
    const hint = picking && c.hint === g.dataset.tool;
    g.classList.toggle('on', picking && (!c.hint || hint)); g.classList.toggle('hint', hint);
    g.classList.toggle('held', g.dataset.tool === c.held);
  });
  document.getElementById('cfDry').classList.toggle('on', k === 'dry' && !c.loaded);
  document.getElementById('cfWet').classList.toggle('on', k === 'wet' && !c.loaded);
  document.getElementById('cfBowl').classList.toggle('on', (k && c.loaded) || (c.phase === 'stir' && c.mix === 0 && !c.stir));
  document.getElementById('cfFront').style.opacity = c.phase === 'stir' ? 1 : 0;
  if (c.anim || c.drag || c.stir) return cfArrow(null);
  if (k && !c.loaded) cfArrow([c.pos[0], c.pos[1] - 40], [CF_SRC[k].at[0], CF_SRC[k].at[1] - 20]);
  else if (k) cfArrow([c.pos[0], c.pos[1] - 40], [CF_BOWL.x + 130, CF_BOWL.y + 10]);
  else cfArrow(null);
}

  const t = CF_TOOLS[id], el = document.querySelector(`#cf-${id} .cf-move`);

  el.style.transform = `translate(${x - t.x}px, ${y - CF_HOME_Y}px) rotate(${rot}deg)`;
  if (cf && id === cf.held) cf.pos = [x, y];
}

  const l = document.querySelector(`#cf-${id} .cf-load`);

function cfPickTool(id) {
  const c = cf, dry = c.phase === 'pickDry';
  if (c.anim || !(dry || c.phase === 'pickWet')) return;
  const r = dry ? c.dry : c.wet;
  if (id !== r.tool) {   // not this one: it shakes, and the right one lights up
    const w = document.querySelector(`#cf-${id} .cf-wig`);
    w.classList.remove('shake'); void w.getBBox(); w.classList.add('shake');
    c.hint = r.tool; cfUpdate();
    return;
  }
  c.held = id; c.count = 0; c.hint = null; c.loaded = false; c.phase = dry ? 'scoopDry' : 'scoopWet';
  cfMove(id, ...CF_HELD, 0, 450);
  c.anim = true; cfUpdate();
  setTimeout(() => { c.anim = false; cfUpdate(); }, 460);
}
function cfDip() {   // into the open food: down, a little scoop, and back up full
  const c = cf, k = cfSrcKey(), S = CF_SRC[k], id = c.held;
  c.anim = true; cfUpdate();
  cfMove(id, ...S.at, 0, 300);
  setTimeout(() => cfMove(id, ...S.dip, 18, 260), 320);
  setTimeout(() => { cfLoad(id, k === 'wet', true); cfMove(id, ...S.rest, 0, 320); }, 620);
  setTimeout(() => { c.anim = false; c.loaded = true; cfUpdate(); }, 960);
}
function cfTip() {   // over the bowl, tip it in
  const c = cf, wet = cfSrcKey() === 'wet', r = wet ? c.wet : c.dry, id = c.held, n = r.tool === 'cup' ? 1 : r.n;
  c.anim = true; cfUpdate();
  cfMove(id, CF_BOWL.x + 150, CF_BOWL.y - 30, 0, 300);
  setTimeout(() => cfMove(id, CF_BOWL.x + 160, CF_BOWL.y - 10, -60, 220), 320);
  setTimeout(() => {
    cfLoad(id, wet, false); c.count++; c.loaded = false;
    const k = Math.min(1, .45 + .55 * c.count / n);
    document.getElementById(wet ? 'cfWetIn' : 'cfDryIn').style.transform = `scale(${k})`;
    if (!wet) document.getElementById('cfKib').insertAdjacentHTML('beforeend', [0, 1, 2].map(i => {
      const f = cfFood(), x = f.cx + ((c.count * 37 + i * 29) % 90 - 45) * k, y = f.cy + ((c.count * 13 + i * 11) % 20 - 10) * k;
      return `<circle cx="${x}" cy="${y}" r="6" fill="#9a6a42"/>`; }).join(''));
    cfMove(id, ...CF_HELD, 0, 260);
    setTimeout(() => {
      c.anim = false;
      if (c.count >= n) {
        if (!wet) { c.phase = 'pickWet'; c.held = null; c.count = 0; cfMove(id, CF_TOOLS[id].x, CF_HOME_Y, 0, 420); }   // back in its place
        else { c.phase = 'stir'; cfMove(id, ...cfStirAt(0, 0), 0, 400); }                                                // down into the food
      }
      cfUpdate();
    }, 280);
  }, 560);
}
function cfStirAt(dx, dy) {   // where the spoon sits while stirring: its tip down in the food, behind the bowl's front
  const f = cfFood();
  return [f.cx + Math.max(-f.rx * .62, Math.min(f.rx * .62, dx)), f.cy + 15 + Math.max(0, Math.min(f.ry * .5, dy))];
}
function cfStirBy(d) {   // mixing: the two foods turn into one
  const c = cf;
  c.mix = Math.min(1, c.mix + d);
  document.getElementById('cfBowl').classList.remove('on');
  document.getElementById('cfMixIn').setAttribute('opacity', c.mix);
  ['cfDryIn', 'cfWetIn', 'cfKib'].forEach(i => { document.getElementById(i).style.opacity = 1 - c.mix; });
  if (c.mix >= 1 && c.phase === 'stir') {
    c.phase = 'done'; c.stir = null;
    const id = c.held; cfMove(id, ...CF_HELD, 0, 300); c.held = null;
    setTimeout(() => cfMove(id, CF_TOOLS[id].x, CF_HOME_Y, 0, 420), 320);   // up out of the food, then back in its place
    cfUpdate();
    const fx = document.createElementNS('http://www.w3.org/2000/svg', 'g'); document.getElementById('msStage').appendChild(fx);
    const f = cfFood(); starBurst(f.cx, f.cy, fx);
    setTimeout(() => { closeOverlay(document.getElementById('measure')); cf = null; feedPet(); }, 1500);
  }
}
function cfTap(el) {   // taps (and Enter) still work: tap a tool to pick it, the food to scoop, the bowl to tip it in or stir
  if (!cf || cf.anim || cf.drag) return;
  const tool = el.closest('.cf-tool'), src = el.closest('.cf-src'), k = cfSrcKey();
  if (tool && !tool.classList.contains('held')) return cfPickTool(tool.dataset.tool);
  if (k && !cf.loaded && src && src.id === (k === 'dry' ? 'cfDry' : 'cfWet')) return cfDip();
  if (k && cf.loaded && src && src.id === 'cfBowl') return cfTip();
  if ((src && src.id === 'cfBowl' || tool) && cf.phase === 'stir' && performance.now() - (cf.tapped || 0) > 400) cfStirTap();
}
function cfStirTap() {   // a tap on the bowl stirs a little
  if (!cf || !cf.held) return;
  cf.tapped = performance.now();
  const id = cf.held;
  cfMove(id, ...cfStirAt(-40, 6), 14, 150); setTimeout(() => cf && cf.held && cfMove(id, ...cfStirAt(40, 10), -14, 150), 160);
  setTimeout(() => cf && cf.held && cfMove(id, ...cfStirAt(0, 0), 0, 150), 320);
  cfStirBy(.25);
}

function jumpPet(from, to, ms_, done) {   // a cat leap: crouch and wiggle, spring up along an arc (stretched out), land with a little squash
  const g = document.querySelector('.petg'), t0 = performance.now(), crouch = 380, land = 220, lean = to[0] < from[0] ? -1 : 1;
  g.style.transition = 'none'; g.style.transformOrigin = `${RUG.cx}px ${RUG.bottom}px`;
  const put = (x, y, s, sx, sy, rot) => { g.style.transform = `translate(${x}px, ${y}px) scale(${s * sx}, ${s * sy}) rotate(${rot}deg)`; };
  const step = t => {
    const e = t - t0;
    if (e < crouch) {   // down low, with a little side to side wiggle, leaning toward where it will jump
      const k = e / crouch, low = Math.sin(Math.min(1, k * 1.6) * Math.PI / 2);
      put(from[0] + Math.sin(k * Math.PI * 4) * 4, from[1], from[2], 1 + .06 * low, 1 - .1 * low, lean * 3 * low);
    } else if (e < crouch + ms_) {   // in the air: long and stretched in the middle, tipped the way it is going
      const k = (e - crouch) / ms_, up = 4 * 170 * k * (1 - k), str = Math.sin(k * Math.PI);
      const x = from[0] + (to[0] - from[0]) * k, y = from[1] + (to[1] - from[1]) * k - up, sc = from[2] + (to[2] - from[2]) * k;
      put(x, y, sc, 1 - .05 * str, 1 + .09 * str, lean * 8 * (1 - 2 * k));
    } else if (e < crouch + ms_ + land) {   // landing: squash, then back to normal
      const k = (e - crouch - ms_) / land, sq = Math.sin(k * Math.PI);
      put(to[0], to[1], to[2], 1 + .07 * sq, 1 - .1 * sq, 0);
    } else { put(to[0], to[1], to[2], 1, 1, 0); return done(); }
    requestAnimationFrame(step);
  };
  requestAnimationFrame(step);
}
function swats(n, done) {   // batting at the feather: the paw goes up (play picture) and down (sitting), and the feather flicks away each time
  const fea = document.querySelector('#wand .feather');
  let i = 0;
  const one = () => {
    if (i++ >= n) { petImg('play', RUG.cx, RUG.bottom, rugH()); return done(); }
    petImg('play', RUG.cx, RUG.bottom, rugH());
    if (fea) { fea.style.translate = `${i % 2 ? 14 : -10}px -16px`; setTimeout(() => { fea.style.translate = '0 0'; }, 170); }
    setTimeout(() => { petImg('sit', RUG.cx, RUG.bottom, rugH()); setTimeout(one, 230 + (i % 2) * 90); }, 260);
  };
  one();
}
function playAtTree(fromDx, fromDy) {
  if (busy || !TASKS[task] || TASKS[task].icon !== 'feather') return;
  busy = true;
  const wand = document.getElementById('wand');
  wand.classList.remove('drag');
  room.querySelector('[data-t="tree"]').classList.remove('on');
  guideEl.classList.add('hide');
  // the wand glides up so the feather dangles just above the bed, and wiggles
  const tx = TREE_BED.cx - (WAND.x + FEATHER[0]) + 46, ty = TREE_BED.bottom - 217 - (WAND.y + FEATHER[1]);   // just above the paw she lifts
  if (fromDx !== undefined) wandMove(fromDx, fromDy, 0);
  requestAnimationFrame(() => wandMove(tx, ty, 500));
  setTimeout(() => wand.classList.add('wiggle'), 450);
  const bed = [TREE_BED.cx - RUG.cx, TREE_BED.bottom - RUG.bottom, TREE_BED.k];
  setTimeout(() => {   // up the tree after it, paws tucked in (the sitting picture)
    petImg('sit', RUG.cx, RUG.bottom, rugH());
    jumpPet([0, 0, 1], bed, 620, () => {
      document.getElementById('treeFront').style.opacity = 1;
      swats(5, () => setTimeout(() => {   // back down, happy
        document.getElementById('treeFront').style.opacity = 0;
        petImg('sit', RUG.cx, RUG.bottom, rugH());
        wand.classList.remove('wiggle'); wand.style.transition = 'opacity .4s'; wand.style.opacity = 0;
        jumpPet(bed, [0, 0, 1], 620, playDone);
      }, 350));
    });
  }, 700);
}
