function svgPoint(e) {
  const p = room.createSVGPoint(); p.x = e.clientX; p.y = e.clientY;

let drag = null;
room.addEventListener('pointerdown', e => {
  const g = e.target.closest('.petg.drag');
  if (!g || busy) return;

  room.setPointerCapture(e.pointerId);
});
room.addEventListener('pointermove', e => {
  if (!drag) return;
  const p = svgPoint(e), dx = p.x - drag.start.x, dy = p.y - drag.start.y;
  if (Math.hypot(dx, dy) > 8) drag.moved = true;
  petMove(dx, dy, 1, 0);

});

  if (!drag) return;

function cupInside(c) {   // the glass's inside from the bottom up to c cups (it is a little wider at the top)
  const y = cupY(c), t = (CUP.y0 - y) / (CUP.y0 - CUP.y1);

}
function kibbleDots(c) {

  }
  return s;
}

function startPour(e) {
  if (!ms || ms.step < 1 || ms.done || ms.pouring) return;
  if (e) e.preventDefault();
  ms.pouring = true; ms.last = performance.now();
  ready(false);
  document.getElementById('msBin').classList.add('pour');
  document.getElementById('msStage').classList.add('pouring');
  requestAnimationFrame(pourTick);
}
function pourTick(t) {
  if (!ms || !ms.pouring) return;
  setCup(ms.amount + POUR_RATE * (t - ms.last) / 1000); ms.last = t;
  requestAnimationFrame(pourTick);
}
function stopPour() {
  if (!ms || !ms.pouring) return;
  ms.pouring = false;
  document.getElementById('msBin').classList.remove('pour');
  document.getElementById('msStage').classList.remove('pouring');
  judge();
}
// is the food at the line? Only then does the bowl show up.
function judge() {
  const goal = CUPS[pet.food], diff = ms.amount - pet.food;
  if (ms.amount <= 0) { ready(false); msButtons(null); msSay(`Hold the food bin to pour, and let go at the ${goal} line.`); return; }
  if (Math.abs(diff) <= CLOSE_ENOUGH) {
    ready(true); msButtons(null);
    msSay(`Just right! Now drag the cup to the bowl and hold it there to pour.`);
  } else if (diff < 0) {
    ready(false); msButtons(null);   // no spoon when there is too little
    msSay(`A little more! Pour up to the ${goal} line.`);
  } else {
    ready(false); msButtons(null, 'on');
    msSay(`Too much! Take some out with the spoon.`);
  }
}

function poseBox(pose, cx, bottom, sitH) {
  const P = POSES[pose], k = sitH / 560;
  return { x: cx - P.w * k / 2 + P.dx * k, y: bottom - P.h * k, w: P.w * k, h: P.h * k };

// which parts of the pet can be rubbed: a grid over the picture, keeping only squares where the pet is
function petCells(artKey, box, cols, rows) {
  return new Promise(res => {
    const im = new Image();
    im.onload = () => {
      const c = document.createElement('canvas'); c.width = 60; c.height = 76;
      const g = c.getContext('2d'); g.drawImage(im, 0, 0, 60, 76);
      const d = g.getImageData(0, 0, 60, 76).data, cells = [];
      for (let r = 0; r < rows; r++) for (let q = 0; q < cols; q++) {
        const px = Math.floor((q + .5) / cols * 60), py = Math.floor((r + .5) / rows * 76);
        if (d[(py * 60 + px) * 4 + 3] > 200 && r < rows * .75)   // the bottom part is inside the tub
          cells.push({ x: box.x + (q + .5) / cols * box.w, y: box.y + (r + .5) / rows * box.h, done: false });
      }
      res(cells);
    };
    im.src = ART[artKey];
  });
}

}

function bathStep() {

function nextBathStep() {

}

function rubAt(x, y, reach) {
  const step = BATH_STEPS[bt.step].id, fx = document.getElementById('btFx');
  let hit = false;
  bt.cells.forEach((c, i) => {
    if (c.done || Math.hypot(c.x - x, c.y - y) > reach) return;
    c.done = true; hit = true;
    if (step === 'soap') {
      const jx = (i * 37) % 23 - 11, jy = (i * 53) % 19 - 9, r = 15 + (i * 7) % 9;   // a little uneven, like real bubbles
      fx.insertAdjacentHTML('beforeend', `<g class="bub" data-cell="${i}"><circle cx="${c.x - 8 + jx}" cy="${c.y + jy}" r="${r}" fill="#fff" opacity=".92"/><circle cx="${c.x + 12 + jx}" cy="${c.y - 12 + jy}" r="${r * .6}" fill="#fff" opacity=".92"/><circle cx="${c.x - 12 + jx}" cy="${c.y - 6 + jy}" r="4" fill="#d7eefc"/></g>`);
    } else if (step === 'rinse') {
      fx.querySelectorAll(`.bub[data-cell="${i}"]`).forEach(b => { b.style.opacity = 0; setTimeout(() => b.remove(), 400); });
    } else {
      const col = step === 'towel' ? '#c8f0da' : '#fff6b0';
      fx.insertAdjacentHTML('beforeend', `<circle class="puff" cx="${c.x}" cy="${c.y}" r="16" fill="${col}"/>`);
    }
  });
  if (!hit) return;
  const part = bt.cells.filter(c => c.done).length / bt.cells.length;
  if (step === 'soap') { const st = document.getElementById('btStink'); if (st) st.style.opacity = Math.max(0, 1 - part / RUB_DONE); }
  if (!bt.ready && part >= RUB_DONE) stepReady(step);
}

function stepReady(step) {
  bt.ready = true;
  if (step === 'soap') { const st = document.getElementById('btStink'); if (st) st.remove(); }
  if (step === 'rinse') setDogPose('bath');
  if (step === 'towel') setDogPose('sit');
  if (step === 'brush') { dropTool(); return bathDone(); }
  bathStep();
}
function setDogPose(pose) {
  document.getElementById('btDog').outerHTML = poseImg(pose);
}

function bathDone() {
  bt.step = BATH_STEPS.length; bt.ready = false; bathStep();
  setDogPose('done');
  document.getElementById('btSay').textContent = `${pet.name} is all clean!`;
  const b = poseBox('done', TUB_DOG.cx, TUB_DOG.bottom, TUB_DOG.h), cx = b.x + b.w / 2, cy = b.y + b.h * .35;
  let stars = '';
  for (let i = 0; i < 8; i++) {   // one small burst of stars, once
    const a = i / 8 * Math.PI * 2;
    stars += `<polygon points="${starPts(cx, cy, 22, 9)}" fill="#ffd34d" style="--bx:${Math.cos(a) * 190}px;--by:${Math.sin(a) * 150}px"/>`;
  }
  document.getElementById('btFx').innerHTML = `<g class="burst">${stars}</g>`;
  setTimeout(() => {
    closeOverlay(document.getElementById('bath'));
    // back on the rug, happy
    const img = document.querySelector('.petg image'), rb = poseBox('done', RUG.cx, RUG.bottom, RUG.h);
    img.setAttribute('href', ART[pet.done]); img.setAttribute('x', rb.x); img.setAttribute('y', rb.y); img.setAttribute('width', rb.w); img.setAttribute('height', rb.h);
    busy = false; setTask(4);
    document.getElementById('adTitle').textContent = `${pet.name} is all done!`;
    document.getElementById('adSay').textContent = `You weighed, fed and bathed ${pet.name}. Great job!`;
    setTimeout(() => openOverlay(document.getElementById('allDone')), 700);
  }, 1600);
}
function starPts(cx, cy, R, r) {
  let p = [];
  for (let i = 0; i < 10; i++) { const a = -Math.PI / 2 + i * Math.PI / 5, rr = i % 2 ? r : R; p.push(`${cx + Math.cos(a) * rr},${cy + Math.sin(a) * rr}`); }
  return p.join(' ');
}
